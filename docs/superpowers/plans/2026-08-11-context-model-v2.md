# Context Model v2 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Give `.wondev/` somewhere to put the three things a production agent-context vault needs and wondev cannot currently express — skill attachments, freshness metadata, and a token-priced memory index — and fix a data-loss bug found while planning it.

**Architecture:** All additive. `MemoryDoc` and `Skill` gain optional fields; a new `src/core/render/index-doc.ts` renders the index as an ordinary `RenderedFile` in `region` mode, which inherits manifest tracking, drift detection and `clean` from the existing writer with no new plumbing. `SOURCE_SCHEMA_VERSION` stays at 1.

**Tech Stack:** TypeScript (ESM, `node:` builtins), `yaml` as the single runtime dependency, vitest, golden-file tests.

## Global Constraints

- **No new runtime dependencies.** `yaml` is the only one and stays the only one.
- **`SOURCE_SCHEMA_VERSION` stays 1.** Nothing authored must change; a bump would make `assertSchemaCurrent` refuse every existing project for a migration with no work to do.
- **No default context budget.** Enforcement only when `index.budget` is explicitly set. A default would fail `check` in every project that upgrades above a number wondev picked.
- **Attachments are `.md` only.** `wholeContent()` prepends an HTML comment to every generated file, which corrupts non-markdown.
- **Attachments are never inlined into flat targets.** Pointer only.
- Every path from config is checked with `isInsideRoot` before reaching the writer.
- Renderers stay pure: same `Project` in, same string out. That is what makes golden files meaningful.
- Task 1 is a PATCH release (`0.1.2`); Tasks 2–10 together are MINOR (`0.2.0`).

/**
 * A deliberate estimate, not a tokenizer.
 *
 * wondev ships one runtime dependency and a real tokenizer would be a second, plus a model
 * choice that goes stale. Four characters per token is the usual rule of thumb for English
 * prose, and the index renders the number behind a `≈` that already promises approximation.

export function estimateTokens(text: string): number {
  return Math.ceil(text.length / 4);
}

export function formatTokens(n: number): string {
  return n < 1000 ? String(n) : `${(n / 1000).toFixed(1)}k`;
}

  if (dir === null) return [];
  const rel = (await listFilesRecursive(dir)).filter((f) => toPosix(f) !== 'SKILL.md');

        file: sourcePath,

      continue;
    }

```ts
function readIndex(value: unknown): IndexConfig | undefined {
  if (value === undefined || value === null) return undefined;
  if (typeof value !== 'object' || Array.isArray(value)) {
    throw new WondevError(`${WONDEV_DIR}/${CONFIG_FILE}: "index" must be a mapping.`);
  }
  const obj = value as Record<string, unknown>;
  const file = asString(obj['file']);
  if (!file) throw new WondevError(`${WONDEV_DIR}/${CONFIG_FILE}: "index.file" is required.`);
  if (!isInsideRoot(file)) {
    throw new WondevError(
      `${WONDEV_DIR}/${CONFIG_FILE}: "index.file" must be a relative path inside the project (got "${file}").`,
    );
  }

  const out: IndexConfig = { file, columns: [] };

  const budget = obj['budget'];
  if (budget !== undefined && budget !== null) {
    if (typeof budget !== 'number' || !Number.isInteger(budget) || budget <= 0) {
      throw new WondevError(`${WONDEV_DIR}/${CONFIG_FILE}: "index.budget" must be a positive integer.`);
    }
    out.budget = budget;
  }

  const columns = obj['columns'];
  if (columns !== undefined && columns !== null) {
    if (!Array.isArray(columns)) {
      throw new WondevError(`${WONDEV_DIR}/${CONFIG_FILE}: "index.columns" must be a list.`);
    }
    for (const raw of columns) {
      const c = raw as Record<string, unknown>;
      const key = asString(c?.['key']);
      const label = asString(c?.['label']);
      if (!key || !label) {
        throw new WondevError(`${WONDEV_DIR}/${CONFIG_FILE}: each index column needs "key" and "label".`);
      }
      out.columns.push({ key, label });
    }
  }
  return out;
}
```

```ts
import type { IndexConfig } from '../config.js';
import type { MemoryDoc, Project, RenderedFile } from '../model.js';
import { estimateTokens, formatTokens } from '../../util/tokens.js';

/** A `|` inside a cell would end it early and shift every column after it. */
function cell(value: unknown): string {
  if (value === undefined || value === null) return '';
  return String(value).replace(/\|/g, '\\|').replace(/\r?\n/g, ' ').trim();
}

export function docTokens(doc: MemoryDoc): number {
  return estimateTokens(doc.body);
}

export function alwaysOnTokens(project: Project): number {
  return project.memory.filter((d) => d.always).reduce((n, d) => n + docTokens(d), 0);
}

function table(docs: MemoryDoc[], showChecked: boolean, columns: IndexConfig['columns']): string {
  const head = ['Note', '≈tok', ...(showChecked ? ['Checked'] : []), 'When to read',
    ...columns.map((c) => c.label)];
  const rows = docs.map((d) => [
    `[[${d.slug}]]`,
    formatTokens(docTokens(d)),
    ...(showChecked ? [d.verified ? `✓ ${d.verified}` : ''] : []),
    cell(d.description),
    ...columns.map((c) => cell(d.extra[c.key])),
  ]);
  return [
    `| ${head.join(' | ')} |`,
    `| ${head.map(() => '---').join(' | ')} |`,
    ...rows.map((r) => `| ${r.join(' | ')} |`),
  ].join('\n');
}

export function renderIndex(project: Project, config: IndexConfig): RenderedFile {
  const showChecked = project.memory.some((d) => d.verified !== undefined);
  const always = project.memory.filter((d) => d.always);
  const onDemand = project.memory.filter((d) => !d.always);

  const out: string[] = ['# Memory index'];

  if (always.length > 0) {
    out.push('## Always loaded');
    out.push("Injected into every target's context on every turn.");
    out.push(table(always, showChecked, config.columns));
    const total = formatTokens(alwaysOnTokens(project));
    out.push(
      config.budget === undefined
        ? `**Always-on total: ≈${total}**`
        : `**Always-on total: ≈${total}** (budget ${formatTokens(config.budget)})`,
    );
  }

  if (onDemand.length > 0) {
    out.push('## On demand');
    out.push('Load only when the trigger matches.');
    out.push(table(onDemand, showChecked, config.columns));
  }

  return { path: config.file, content: `${out.join('\n\n').trim()}\n`, mode: 'region' };
}
```
